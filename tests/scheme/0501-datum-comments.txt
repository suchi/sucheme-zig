# スライス 5a: データコメント `#;`(R7RS 2.2、7.1.1 の <comment>)。続く1つのデータを
# 読んで捨てる。

(list 1 #;2 3)
(1 3)

(list 1 #; (car '()) 2)
(1 2)

# 捨てたデータは評価しない。
#;(car '()) 'ok
ok

# `#;` に続く空白の中の `#;` は、後ろのデータを先に捨てる。
(list #; #; 1 2 3)
(3)

# 略記はデータの一部なので、一緒に捨てる。
(list #;'a 'b)
(b)

'(a . #;b c)
(a . c)

'#(1 #;2 3)
#(1 3)

# 捨てるデータしか無い入力は、何も評価しない。
#;1
<no output>

# データが無いところで終わったら、誤りである。
(list 1 #;)
error: datum comment needs a datum

(list 1 #;
error: unexpected end of input

# 捨てるデータも、読めるデータでなければならない。
#;#z 1
error: unreadable token
