# Slices 3s and 4g-2: the integer procedures of R7RS 6.2.6. The division procedures and
# abs are native; the rest are written in Scheme over them. The examples are those of
# R7RS 6.2.6 (docs/spec/). リーダは flonum の綴りを読めない(スライス 4r)ので、不正確な
# 数は `inexact` で作る。
#
# 1ケースは1行で、エラーはその行の残りを打ち切る。

# 最初の失敗するテスト。
(list (zero? 0) (zero? 3) (positive? 3) (positive? -3) (positive? 0) (negative? -3) (negative? 0))
(#t #f #t #f #f #t #f)

(list (odd? 3) (odd? -3) (odd? 0) (even? 0) (even? -4) (even? 7))
(#t #t #f #t #t #f)

# abs と square。
(list (abs -7) (abs 7) (abs 0) (abs -1) (square 42) (square -3))
(7 7 0 1 1764 9)

# floor/ と truncate/ は2つの値を返す。
(call-with-values (lambda () (floor/ 5 2)) list)
(2 1)

(call-with-values (lambda () (floor/ -5 2)) list)
(-3 1)

(call-with-values (lambda () (floor/ 5 -2)) list)
(-3 -1)

(call-with-values (lambda () (floor/ -5 -2)) list)
(2 -1)

(call-with-values (lambda () (truncate/ 5 2)) list)
(2 1)

(call-with-values (lambda () (truncate/ -5 2)) list)
(-2 -1)

(call-with-values (lambda () (truncate/ 5 -2)) list)
(-2 1)

(call-with-values (lambda () (truncate/ -5 -2)) list)
(2 -1)

# 1つの値を返す版は、2つの値の版と同じ商と余りになる。
(list (floor-quotient -5 2) (floor-remainder -5 2) (floor-quotient 5 -2) (floor-remainder 5 -2) (floor-quotient 6 3) (floor-remainder 6 3))
(-3 1 -3 -1 2 0)

(list (truncate-quotient -5 2) (truncate-remainder -5 2) (truncate-quotient 5 -2) (truncate-remainder 5 -2))
(-2 -1 -2 1)

# gcd と lcm は、0個以上の引数を取り、結果は負にならない。
(list (gcd 32 -36) (gcd) (gcd 0 5) (gcd 0 0) (gcd 7) (gcd -7) (gcd 3 -1) (gcd -1) (gcd 12 18 8))
(4 0 5 0 7 7 1 1 2)

(list (lcm 32 -36) (lcm) (lcm 0 5) (lcm 0 0) (lcm -7) (lcm 2 3 4))
(288 1 0 0 7 12)

# 結果が固定長の整数に収まるなら、途中の値もあふれない。引数の順序で結果は変わらない。
(list (floor-quotient -9223372036854775808 3) (floor-quotient 9223372036854775807 -2) (gcd -9223372036854775808 6) (gcd 6 -9223372036854775808))
(-3074457345618258603 -4611686018427387904 2 2)

(call-with-values (lambda () (floor/ -9223372036854775808 3)) list)
(-3074457345618258603 1)

# 真の値が fixnum に収まらないときは、bignum になる(4f から。以前は誤りだった)。
(gcd -9223372036854775808)
9223372036854775808

# (scheme base) からエクスポートされ、import の集合で選べる。
(import (only (scheme base) even? floor/)) (call-with-values (lambda () (floor/ 7 (if (even? 2) 2 3))) list)
(3 1)

(truncate-quotient 1 0)

error: division by zero in truncate-quotient 1 0

# 不正確な整数と bignum。原文の例: (lcm 32.0 -36) は 288.0、(truncate/ -5.0 -2) は 2.0 と -1.0。
(list (lcm (inexact 32) -36) (gcd (inexact 32) -36) (abs (inexact -7)))
(288.0 4.0 7.0)

(call-with-values (lambda () (truncate/ (inexact -5) -2)) list)
(2.0 -1.0)

(call-with-values (lambda () (floor/ (inexact 5) -2)) list)
(-3.0 -1.0)
