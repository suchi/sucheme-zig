# 深さ数千の入れ子の `equal?`。car の方向の入れ子は作業スタックを深くする。
(define (nest n acc) (if (= n 0) acc (nest (- n 1) (list acc)))) (equal? (nest 3000 '()) (nest 3000 '()))
#t

(define (nest n acc) (if (= n 0) acc (nest (- n 1) (list acc)))) (equal? (nest 3000 1) (nest 3000 2))

# 深さの違う入れ子は、短いほうの底で食い違う。

# ベクタの入れ子。要素が2つ以上のベクタは作業スタックに積まれる。

(define (vnest n acc) (if (= n 0) acc (vnest (- n 1) (vector acc 0)))) (equal? (vnest 3000 1) (vnest 3000 2))
