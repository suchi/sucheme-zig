# スライス 3d: 文字列のリテラルと `symbol->string` の結果は定数で、書き込む手続きは
# 誤りとして断る(R7RS 3.4、4.1.2、6.7)。`string-copy` や `string-append` などが作る
# 文字列は、新しく作るので、定数ではなく、書き込める。
# 各ケースは新しいインタプリタで走るので、定義は使う式と同じ行に置く。
#
# 原文(R7RS 6.7)の例:
#   (define (f) (make-string 3 #\*))
#   (define (g) "***")
#   (string-set! (f) 0 #\?)   => unspecified
#   (string-set! (g) 0 #\?)   => error
#   (string-set! (symbol->string 'immutable) 0 #\?) => error

(define (f) (make-string 3 #\*)) (define (g) "***") (string-set! (f) 0 #\?)
<no output>

(define (f) (make-string 3 #\*)) (define (g) "***") (string-set! (g) 0 #\?)
error: cannot modify a constant in string-set! "***" 0 #\?

(string-set! (symbol->string 'immutable) 0 #\?)
error: cannot modify a constant in string-set! "immutable" 0 #\?

# リテラルは、式の中でも、データの中でも定数である。
(string-set! "abc" 0 #\x)
error: cannot modify a constant in string-set! "abc" 0 #\x

(string-set! (car '("abc" "def")) 0 #\x)
error: cannot modify a constant in string-set! "abc" 0 #\x

(string-set! (vector-ref #("abc" 1) 0) 0 #\x)
error: cannot modify a constant in string-set! "abc" 0 #\x

(string-set! (cdr '(1 . "abc")) 0 #\x)
error: cannot modify a constant in string-set! "abc" 0 #\x

(define s "abc") (string-set! s 0 #\x)
error: cannot modify a constant in string-set! "abc" 0 #\x

# `string-fill!` と `string-copy!` も同じである。
(string-fill! "abc" #\x)
error: cannot modify a constant in string-fill! "abc" #\x

(string-copy! "abc" 0 "xy")
error: cannot modify a constant in string-copy! "abc" 0 "xy"

(string-fill! (symbol->string 'abc) #\x)
error: cannot modify a constant in string-fill! "abc" #\x

(string-copy! (symbol->string 'abc) 0 "xy")
error: cannot modify a constant in string-copy! "abc" 0 "xy"

# 書き込む範囲が空でも、定数への書き込みである。
(string-fill! "" #\x)
error: cannot modify a constant in string-fill! "" #\x

(string-fill! "abc" #\x 1 1)
error: cannot modify a constant in string-fill! "abc" #\x 1 1

(string-copy! "abc" 3 "")
error: cannot modify a constant in string-copy! "abc" 3 ""

# 同じ文字を書く場合も、断る。
(string-set! "abc" 0 #\a)
error: cannot modify a constant in string-set! "abc" 0 #\a

# 書き込む位置が範囲の外でも、型が正しければ、定数であることが先に報告される。
(string-set! "abc" 10 #\x)
error: cannot modify a constant in string-set! "abc" 10 #\x

# 型の誤りは、定数の誤りより先に報告される。
(string-set! "abc" 0 1)
error: wrong type of argument to string-set! "abc" 0 1

# 定数の文字列を、書き込み先でなく、読む側にする手続きは、通る。
(string-length "abc")
3

(string-append "ab" "cd")
"abcd"

(define t (string-copy "abc")) (string-copy! t 1 "xy") t
"axy"

# `string-copy` と `substring` は、定数の写しを作る。写しは定数ではない。
(define t (string-copy "abc")) (string-set! t 0 #\x) t
"xbc"

(define t (substring "abc" 1 3)) (string-set! t 0 #\x) t
"xc"

(define s "abc") (define t (string-copy s)) (string-set! t 0 #\x) (list s t)
("abc" "xbc")

(define t (string-copy (symbol->string 'abc))) (string-set! t 0 #\x) t
"xbc"

(define t (string-copy "")) (string-fill! t #\x) t
""

# `string-append` の結果は新しく、定数ではない。1つの引数でも、同じ文字列を返さない。
(define s "abc") (define t (string-append s)) (list (eq? s t) (equal? s t))
(#f #t)

(define t (string-append "ab" "cd")) (string-set! t 0 #\x) t
"xbcd"

(define t (string-append "abc")) (string-set! t 0 #\x) t
"xbc"

# 手続きが作る文字列は定数ではない。
(define t (make-string 3 #\a)) (string-set! t 0 #\x) t
"xaa"

(define t (string #\a #\b)) (string-set! t 0 #\x) t
"xb"

(define t (list->string (list #\a #\b))) (string-set! t 0 #\x) t
"xb"

(define t (vector->string #(#\a #\b))) (string-set! t 0 #\x) t
"xb"

# `symbol->string` は、呼ぶたびに新しい文字列を返し、どれも定数である。名前の表は変わらない。
(define s (symbol->string 'abc)) (list (eq? s (symbol->string 'abc)) (equal? s "abc") (eq? 'abc (string->symbol s)))
(#f #t #t)

(define s (symbol->string 'abc)) (call/cc (lambda (k) (with-exception-handler (lambda (e) (k 'refused)) (lambda () (string-set! s 0 #\x))))) (list s (eq? 'abc (string->symbol "abc")))
refused
("abc" #t)

# `string->symbol` に渡した文字列は、定数にならない。
(define s (make-string 2 #\a)) (string->symbol s) (string-set! s 0 #\b) s
aa
"ba"

# 定数への書き込みを断ったあとも、文字列は元のままである。
(define s "abc") (call/cc (lambda (k) (with-exception-handler (lambda (e) (k 'refused)) (lambda () (string-set! s 1 #\x))))) s
refused
"abc"

# 印は、GC のあとも残る。
(define s "abc") (define (spin n) (if (= n 0) 'ok (spin (- n 1)))) (spin 20000) (string-set! s 0 #\x)
ok
error: cannot modify a constant in string-set! "abc" 0 #\x

# マクロが出力する文字列のリテラルも定数である。
(define-syntax m (syntax-rules () ((_) "abc"))) (string-set! (m) 0 #\x)
error: cannot modify a constant in string-set! "abc" 0 #\x
