# 黙って別の意味に読まず、まだ扱えないと報告する。それらの行は、入れるコミットが書き換える。

(define-syntax m (syntax-rules () ((_ x) 1))) (m)
error: bad syntax: no rule of m matches the form

(define-syntax m (syntax-rules () ((_ a a) 1)))
error: bad syntax-rules: duplicate pattern variable: a

(define-syntax m (syntax-rules () ((_ a ...) a)))
error: bad syntax-rules: a needs as many ellipses in the template as in its pattern

(define-syntax m (syntax-rules () ((_ a) (a ...))))
error: bad syntax-rules: an ellipsis must follow a template with a pattern variable that an ellipsis followed

(define-syntax m (syntax-rules () ((_ ... a) 1)))

(define-syntax m (syntax-rules () ((_ a) ...)))
error: bad syntax-rules: an ellipsis must follow a template

(define-syntax m (syntax-rules () ((_ a) (list ...))))
error: bad syntax-rules: an ellipsis must follow a template with a pattern variable that an ellipsis followed

error: bad syntax: a and b matched different numbers of forms but an ellipsis repeats them together

(define-syntax m)
