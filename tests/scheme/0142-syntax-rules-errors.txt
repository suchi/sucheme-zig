(define-syntax m (syntax-rules () ((_ a a) 1)))

(define-syntax m (syntax-rules () ((_ a ...) a)))

(define-syntax m (syntax-rules () ((_ ... a) 1)))

(define-syntax m (syntax-rules () ((_ a) ...)))
error: 1:1: bad syntax-rules: an ellipsis must follow a template

(define-syntax m (syntax-rules () ((_ a) (list ...))))
error: 1:1: bad syntax-rules: an ellipsis must follow a template with a pattern variable that an ellipsis followed

# 2l が入れる形。
(define-syntax m (syntax-rules () ((_ a ...) (a ... ...))))

(define-syntax m (syntax-rules () ((_ a ... b ...) 1)))

(define-syntax m (syntax-rules () ((_ a ... ...) 1)))
error: 1:1: bad syntax-rules: a list pattern can have only one ellipsis

# 内側の変数が深さ2なのに、テンプレートの省略記号が1つだけ。
(define-syntax m (syntax-rules () ((_ (a ...) ...) '(a ...))))
error: 1:1: bad syntax-rules: a needs as many ellipses in the template as in its pattern

(define-syntax m (syntax-rules () ((_ #(a ... b ...)) 1)))
error: 1:1: bad syntax-rules: a list pattern can have only one ellipsis

(define-syntax m (syntax-rules () ((_ #(a ...)) #(a))))
error: 1:1: bad syntax-rules: a needs as many ellipses in the template as in its pattern

(define-syntax m (syntax-rules () ((_ a) #(... a))))
error: 1:1: bad syntax-rules: an ellipsis must follow a template

(define-syntax m (syntax-rules :::))
