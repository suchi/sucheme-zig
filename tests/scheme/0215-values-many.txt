# スライス 3b: 個数の多い複数値。`call-with-values` の受け手の引数の鎖は、複数値の
# オブジェクトが届いたとき、継続のフレームに持たせて、境目を挟む複数のステップで並べる。
# 1ステップで並べる個数(src/machine.zig の `spread_chunk`)の前後の個数で、値と順序が
# 保たれることを確かめる。各ケースは新しいインタプリタで走るので、定義は使う式と
# 同じ行に置く。規模を試す双子は tests/scheme/0217-apply-values-scale-small.txt と
# tests/scheme-slow/0019-apply-values-scale.txt にある。

(define (build n acc) (if (= n 0) acc (build (- n 1) (cons n acc)))) (define xs (build 200 '())) (call-with-values (lambda () (apply values xs)) (lambda (a b . r) (list a b (length r))))

# 受け手の個数が合わなければ、ふつうの手続き呼び出しと同じ誤りになる。

# 受け手は末尾呼び出しで、受け手の戻りが call-with-values の値になる。
(define (build n acc) (if (= n 0) acc (build (- n 1) (cons n acc)))) (+ 1 (call-with-values (lambda () (apply values (build 100 '()))) +))

# 生産者が `values` 自身なら、複数値が複数値のまま、もう一度受け手に渡る。
(define (build n acc) (if (= n 0) acc (build (- n 1) (cons n acc)))) (call-with-values (lambda () (call-with-values (lambda () (apply values (build 70 '()))) values)) +)
2485

# 受け手がまた `call-with-values` を呼ぶ入れ子でも、同じである。
(define (build n acc) (if (= n 0) acc (build (- n 1) (cons n acc)))) (define (rest-sum . xs) (apply + xs)) (call-with-values (lambda () (apply values (build 90 '()))) (lambda args (call-with-values (lambda () (apply values args)) rest-sum)))
4095
