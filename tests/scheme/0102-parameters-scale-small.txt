# tests/scheme-slow/0017-parameters-scale.txt の縮小版。同じ論理を、毎回の門で走る大きさで
# 通す。深く入れ子にした parameterize、その内側からの脱出と再入、読み出しと束縛の繰り返しで、
# GC が何度も走る(実行器は1ステップごとに GC するモードでも走らせる)。

# 500段の入れ子の底で、最も内側の束縛が見え、外に出れば最初の値に戻る。
(define p (make-parameter 0)) (define (nest n thunk) (if (= n 0) (thunk) (parameterize ((p n)) (+ 0 (nest (- n 1) thunk))))) (list (nest 500 (lambda () (+ (p) 1000))) (p))
(1001 0)

# 入れ子の底で捕捉した継続で、脱出して再入する。変換手続きは束縛ごとに1回しか呼ばれず、再入で
# 呼び直されない。再入した底で見える値は、脱出する前と同じである。
(define count 0) (define p (make-parameter 0 (lambda (x) (set! count (+ count 1)) x))) (define k #f) (define round 0) (define seen '()) (define (nest n thunk) (if (= n 0) (thunk) (parameterize ((p n)) (+ 0 (nest (- n 1) thunk))))) (define r (call/cc (lambda (out) (nest 500 (lambda () (call/cc (lambda (c) (set! k c))) (set! seen (cons (p) seen)) (out 'escaped)))))) (set! round (+ round 1)) (if (= round 1) (k #f)) (list r round (p) count seen)
(escaped 1 0 501 (1 1))

# 束縛の内側での読み出しを、繰り返す。
(define p (make-parameter 1)) (define (sum n acc) (if (= n 0) acc (sum (- n 1) (+ acc (p))))) (parameterize ((p 2)) (sum 5000 0))
10000

# 束縛を、繰り返す。1回ごとに、点と、本体のフレームを作って捨てる。
(define p (make-parameter 0)) (define (loop n acc) (if (= n 0) acc (loop (- n 1) (+ acc (parameterize ((p n)) (p)))))) (loop 1000 0)
500500
