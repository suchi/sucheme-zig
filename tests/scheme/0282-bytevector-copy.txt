# スライス 3e: `bytevector-copy` と `bytevector-copy!`(R7RS 6.9)。範囲は、始まりを含み、終わりを
# 含まない。検査の順序は、型、定数かどうか、範囲である。
# 各ケースは新しいインタプリタで走るので、定義は使う式と同じ行に置く。

# `bytevector-copy`: 原文の例。
(define a #u8(1 2 3 4 5)) (bytevector-copy a 2 4)
#u8(3 4)

(bytevector-copy #u8(1 2 3 4 5))
#u8(1 2 3 4 5)

(bytevector-copy #u8(1 2 3 4 5) 2)
#u8(3 4 5)

(bytevector-copy #u8(1 2 3 4 5) 5)
#u8()

(bytevector-copy #u8(1 2 3 4 5) 2 2)
#u8()

(bytevector-copy #u8(1 2 3 4 5) 0 5)
#u8(1 2 3 4 5)

(bytevector-copy #u8())
#u8()

# 写しは新しいバイトベクタで、リテラルの写しも書き換えられる。
(define a #u8(1 8 2 8)) (define b (bytevector-copy a)) (bytevector-u8-set! b 0 3) (list a b)
(#u8(1 8 2 8) #u8(3 8 2 8))

(define a (bytevector 1 2 3)) (eq? a (bytevector-copy a))
#f

(define a (bytevector 1 2 3)) (define b (bytevector-copy a 1)) (bytevector-u8-set! a 1 9) b
#u8(2 3)

(bytevector-copy #u8(1 2 3) 4)
error: 1:1: argument out of range in bytevector-copy #u8(1 2 3) 4

(bytevector-copy #u8(1 2 3) 0 4)
error: 1:1: argument out of range in bytevector-copy #u8(1 2 3) 0 4

(bytevector-copy #u8(1 2 3) 2 1)
error: 1:1: argument out of range in bytevector-copy #u8(1 2 3) 2 1

(bytevector-copy #u8(1 2 3) -1)
error: 1:1: argument out of range in bytevector-copy #u8(1 2 3) -1

(bytevector-copy #u8(1 2 3) 0 -1)
error: 1:1: argument out of range in bytevector-copy #u8(1 2 3) 0 -1

(bytevector-copy #u8(1 2 3) 0 9223372036854775807)
error: 1:1: argument out of range in bytevector-copy #u8(1 2 3) 0 9223372036854775807

(bytevector-copy #u8(1 2 3) 4294967296 4294967296)
error: 1:1: argument out of range in bytevector-copy #u8(1 2 3) 4294967296 4294967296

(bytevector-copy "abc")
error: 1:1: wrong type of argument to bytevector-copy "abc"

(bytevector-copy #u8(1 2 3) #\a)
error: 1:1: wrong type of argument to bytevector-copy #u8(1 2 3) #\a

(bytevector-copy #u8(1 2 3) 0 #\a)
error: 1:1: wrong type of argument to bytevector-copy #u8(1 2 3) 0 #\a

# 型の誤りが、範囲の誤りより先に報告される。
(bytevector-copy #u8(1 2 3) 9 #\a)
error: 1:1: wrong type of argument to bytevector-copy #u8(1 2 3) 9 #\a

# `bytevector-copy!`: 原文の例。
(define a (bytevector 1 2 3 4 5)) (define b (bytevector 10 20 30 40 50)) (bytevector-copy! b 1 a 0 2) b
#u8(10 1 2 40 50)

# 値は返さない。
(bytevector-copy! (bytevector 1 2) 0 #u8(3))
<no output>

# 範囲の省略。
(define b (bytevector 0 0 0 0 0)) (bytevector-copy! b 0 #u8(1 2 3 4 5)) b
#u8(1 2 3 4 5)

(define b (bytevector 0 0 0 0 0)) (bytevector-copy! b 2 #u8(1 2 3 4 5) 3) b
#u8(0 0 4 5 0)

(define b (bytevector 0 0 0 0 0)) (bytevector-copy! b 1 #u8(1 2 3 4 5) 1 3) b
#u8(0 2 3 0 0)

# 何も写さない呼び出しは、末尾の位置からでもよい。
(define b (bytevector 1 2 3)) (bytevector-copy! b 3 #u8(9 9) 1 1) b
#u8(1 2 3)

(define b (bytevector 1 2 3)) (bytevector-copy! b 0 #u8()) b
#u8(1 2 3)

# 残りにちょうど収まる。
(define b (bytevector 1 2 3)) (bytevector-copy! b 1 #u8(8 9)) b
#u8(1 8 9)

# 元が定数でもよい。
(define b (make-bytevector 3 0)) (bytevector-copy! b 0 '#u8(7 8 9)) b
#u8(7 8 9)

# 元と先が同じで範囲が重なるときは、元を一時の領域に写してから書いたように振る舞う。
(define a (bytevector 1 2 3 4 5)) (bytevector-copy! a 1 a 0 4) a
#u8(1 1 2 3 4)

(define a (bytevector 1 2 3 4 5)) (bytevector-copy! a 0 a 1 5) a
#u8(2 3 4 5 5)

(define a (bytevector 1 2 3 4 5)) (bytevector-copy! a 2 a 0 3) a
#u8(1 2 1 2 3)

(define a (bytevector 1 2 3 4 5)) (bytevector-copy! a 0 a 2 5) a
#u8(3 4 5 4 5)

(define a (bytevector 1 2 3)) (bytevector-copy! a 1 a 1) a
#u8(1 2 3)

(define a (bytevector 1 2 3)) (bytevector-copy! a 0 a) a
#u8(1 2 3)

# 写す前に断る。断られたあとの先は、元のままである。
