# スライス 5c: データラベル `#<n>=` と `#<n>#`(R7RS 2.4)。循環と共有を `quote` したデータが、
# `eq?` で同一性を保つ(docs/design/frontend.md の衛生性の検証項目の9)。
# 今のプリンタは循環を検出しないので(プリンタのスライス 5j で直す)、循環したデータは
# 結果として書かず、`eq?` で確かめる。

(let ((x '#0=(a . #0#))) (eq? x (cdr x)))
#t

(let ((x '#0=(a b c . #0#))) (eq? x (cdr (cdr (cdr x)))))
#t

(let ((v '#0=#(1 #0#))) (eq? v (vector-ref v 1)))
#t

# 共有。ラベルを読み終えたあとの参照は、同じオブジェクトである。
(let ((x '(#0=(1) #0#))) (eq? (car x) (cadr x)))
#t

(let ((x '(#0=(1) #0#))) x)
((1) (1))

# 入れ子のラベル。
(let ((x '#0=(#1=(b . #1#) . #0#))) (list (eq? x (cdr x)) (eq? (car x) (cdar x))))
(#t #t)

# 既知の制限: マクロの引数として渡した循環データは、マクロの中で quote されるとしても、
# 読んだ時点ではリテラルの外にあるので、誤りにする(docs/design/frontend.md の「現状(スライス 5c)」)。
(define-syntax m (syntax-rules () ((_ d) 'd))) (let ((x (m #0=(a . #0#)))) (eq? x (cdr x)))
error: circular reference outside a literal

# 循環したリテラルも定数で、変更できない。誤りの説明の irritant も、`write` の形でラベルを
# 付けて書く(スライス 5j)。
(set-car! '#0=(a . #0#) 1)
error: 1:1: cannot modify a constant in set-car! #0=(a . #0#) 1

# 循環したリスト同士の `equal?` は止まる(R7RS 6.1)。
(equal? '#0=(a . #0#) '#1=(a . #1#))
#t

# リテラルの外の循環は誤りである(R7RS 2.4 の例)。リーダが、コードとして読むときに断る。
#1=(begin (display #\x) #1#)
error: circular reference outside a literal

(list 1 #0=(+ 1 #0#))
error: circular reference outside a literal

# quasiquote の中の循環も誤りである(R7RS 2.4)。
`#0=(a ,@#0#)
error: circular reference outside a literal

# quasiquote はベクタの中まで辿るので、ベクタと quote の中の循環も断る。
`#(1 ,#0=(a . #0#))
error: circular reference outside a literal

`(1 '#0=(a . #0#))
error: circular reference outside a literal

# cdr として出会う `(quote x)` は quote の式ではなく、リストの尾部である。
#0=(lambda quote #0#)
error: circular reference outside a literal

# テンプレートとパターンの循環は、マクロを定義する時点で断る。
(define-syntax q (syntax-rules () ((_) '#0=(tmp . #0#))))
error: 1:1: bad syntax-rules: circular template

(define-syntax q (syntax-rules () ((_ #(#0=(a . #0#))) 1)))
error: 1:1: bad syntax-rules: circular pattern

# ラベルの範囲は、最も外側のデータである(R7RS 2.4)。
'#0=a '#0#
a
error: reference to an undefined datum label

'(#0=a #0=b)
error: datum label defined twice

'#0=#0#
error: datum label needs a datum
