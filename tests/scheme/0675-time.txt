# スライス 5n: `current-second`、`current-jiffy`、`jiffies-per-second`(R7RS 6.14)。
# 時計は本物を読むので、値でなく性質を見る(src/platform/clock.zig)。

# jiffy はマイクロ秒である。
(import (scheme time)) (jiffies-per-second)
1000000

# `current-jiffy` は正の正確な整数で、続けて呼んでも減らない。
(import (scheme time)) (let* ((a (current-jiffy)) (b (current-jiffy))) (list (exact-integer? a) (< 0 a) (<= a b)))
(#t #t #t)

# `current-second` は不正確な数で、2017-01-01 の UNIX 時刻(1483228800)より後、2100-01-01(4102444800)より前である。
(import (scheme time)) (let ((s (current-second))) (list (inexact? s) (real? s) (< 1483228800 s 4102444800)))
(#t #t #t)

# R7RS 6.14 の例: 経過時間を秒で測る。
(import (scheme time)) (let ((start (current-jiffy))) (let loop ((i 0)) (if (< i 1000) (loop (+ i 1)))) (let ((elapsed (/ (- (current-jiffy) start) (jiffies-per-second)))) (and (exact? elapsed) (<= 0 elapsed))))
#t

# 引数は取らない。
(import (scheme time)) (current-jiffy 1)
error: wrong number of arguments to current-jiffy 1
