# Slice 2m: the macros of a library. A macro a library defines keeps the meaning of the
# identifiers its template has in the library, wherever it is used, and can be exported,
# renamed and imported like any other binding (R7RS 5.6.1, 4.3).
#
# 1ケースは1行で、エラーはその行の残りを打ち切る。

(define-library (t m) (export swap!) (import (scheme base)) (begin (define-syntax swap! (syntax-rules () ((_ a b) ((lambda (tmp) (set! a b) (set! b tmp)) a)))))) (import (t m)) ((lambda (tmp x) (swap! tmp x) (list tmp x)) 1 2)
(2 1)

# 省略記号は、インポートした `...` で読む。
(define-library (t m) (export m) (import (scheme base)) (begin (define-syntax m (syntax-rules () ((_ x ...) (list x ...)))))) (import (t m)) (m 1 2 3)
(1 2 3)

# テンプレートの識別子は、ライブラリの束縛を指す。利用側が同じ名前を持っていても、
# エクスポートされていない手続きを指せる(参照透明性)。
(define-library (t m) (export twice) (import (scheme base)) (begin (define (double x) (* x 2)) (define-syntax twice (syntax-rules () ((_ e) (double e)))))) (import (t m)) (twice 4)
8

(define-library (t m) (export twice) (import (scheme base)) (begin (define (double x) (* x 2)) (define-syntax twice (syntax-rules () ((_ e) (double e)))))) (import (t m)) (define (double x) 'repl) (list (twice 4) (double 4))
(8 repl)

(define-library (t m) (export twice) (import (scheme base)) (begin (define (double x) (* x 2)) (define-syntax twice (syntax-rules () ((_ e) (double e)))))) (import (t m)) ((lambda (double) (twice 5)) 'local)
10

# 衛生性の検証項目3: 利用側が `if` を再束縛していても、テンプレートの `if` は本来の意味を保つ。
(define-library (t m) (export given) (import (scheme base)) (begin (define-syntax given (syntax-rules () ((_ test body) (if test body #f)))))) (import (t m)) ((lambda (if) (given if (list if))) #t)
(#t)

# ライブラリが `if` をインポートしていなくても、マクロを定義したライブラリの意味で読む。
(define-library (t m) (export given) (import (only (scheme base) define-syntax syntax-rules if)) (begin (define-syntax given (syntax-rules () ((_ test body) (if test body #f)))))) (import (t m)) (define (if a b c) 'repl) (given #t 1)
1

# ライブラリが定義した、マクロを定義するマクロ。
(define-library (t m) (export def-const) (import (scheme base)) (begin (define-syntax def-const (syntax-rules () ((_ name v) (define-syntax name (syntax-rules () ((_) v)))))))) (import (t m)) (def-const five 5) (five)

# REPL は、インポートしたマクロを `define-syntax` で遮蔽できる。ライブラリの定義は変わらない。
(define-library (t m) (export m) (import (scheme base)) (begin (define-syntax m (syntax-rules () ((_) 'library))))) (import (t m)) (define-syntax m (syntax-rules () ((_) 'repl))) (m)
repl

(define-library (t m) (export m) (import (scheme base)) (begin (define-syntax m (syntax-rules () ((_) 'library))))) (define-library (t n) (export u) (import (scheme base) (t m)) (begin (define (u) (m)))) (import (t m)) (define-syntax m (syntax-rules () ((_) 'repl))) (import (t n)) (list (m) (u))
(repl library)

# ライブラリの本体は、インポートしたマクロを `define-syntax` で再定義できない。
(define-library (t m) (export m) (import (scheme base)) (begin (define-syntax m (syntax-rules () ((_) 1))))) (define-library (t n) (export) (import (scheme base) (t m)) (begin (define-syntax m (syntax-rules () ((_) 2)))))
error: bad define-syntax: m is an imported binding and cannot be redefined

(define-library (t m) (export m) (import (scheme base)) (begin (define-syntax m (syntax-rules () ((_) 1))))) (define-library (t n) (export) (import (scheme base) (t m)) (begin (define m 2)))
error: bad define: m is an imported binding and cannot be redefined

(define-library (t m) (export m) (import (scheme base)) (begin (define-syntax m (syntax-rules () ((_) 1))))) (define-library (t n) (export) (import (scheme base) (t m)) (begin (set! m 2)))
error: bad set!: m is a syntactic keyword and cannot be assigned
