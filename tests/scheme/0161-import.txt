# Slice 2m: `import` at the REPL (R7RS 5.2). The REPL imports `(scheme base)` to begin
# with; an import adds bindings, or replaces the binding of the same name. The five
# forms of an import set: a library name, `only`, `except`, `prefix` and `rename`.
#
# 1ケースは1行で、エラーはその行の残りを打ち切る。

# インポートは何も表示しない。
(import (scheme base))
<no output>

(import (only (scheme base) car cdr)) (car '(1 2))
1

# `rename` は別の名前で束縛する。元の名前の束縛は、そのまま残る。特殊形式も改名できる。
(import (rename (scheme base) (if when-true))) (list (when-true #t 1 2) (if #f 1 2))
(1 2)

(import (rename (only (scheme base) car cdr) (car first) (cdr rest))) (list (first '(1 2)) (rest '(1 2)))
(1 (2))

# `rename` は同時に行う。名前の入れ替えができる。
(import (rename (only (scheme base) car cdr) (car cdr) (cdr car))) (list (car '(1 2)) (cdr '(1 2)))
((2) 1)

(import (prefix (only (scheme base) car cdr) c:)) (list (c:car '(1 2)) (c:cdr '(1 2)))
(1 (2))

# `except` は、列挙した名前を除く。REPL がすでに持つ束縛は、除いても残る。
(import (rename (except (scheme base) car) (cdr kdr))) (list (kdr '(1 2)) (car '(3)))
((2) 3)

# 修飾子は入れ子にできる。
(import (prefix (rename (only (scheme base) car) (car head)) p-)) (p-head '(1 2))
1

# 補助構文もインポートできる。改名した名前は、元の名前と同じ補助構文を意味するので、
# リテラルとして照合するとき、どちらの綴りも一致する。
(import (rename (only (scheme base) else) (else otherwise))) (define-syntax pick (syntax-rules (otherwise) ((_ otherwise) 'yes) ((_ x) 'no))) (list (pick else) (pick otherwise) (pick 1))
(yes yes no)

# REPL の定義で隠した特殊形式とプリミティブは、インポートで取り戻せる。
(define if 5) (import (only (scheme base) if)) (if #t 1 2)
1

(define car 5) (import (only (scheme base) car)) (car '(7))
7

(define (car x) 'mine) (define (f) (car '(1))) (import (only (scheme base) car)) (list (f) (car '(1)))
(mine 1)

# 先に展開した式が指す REPL 自身の変数は、インポートで置き換えられても、そのまま働く。
(define n 1) (define (get) n) (import (rename (only (scheme base) car) (car n))) (list (get) (n '(2)))
(1 2)

# 1つの `import` の中の誤りは、その `import` のどの集合も束縛しない。
(import (rename (only (scheme base) car) (car kar)) (only (scheme base) no-such-name))
error: 1:1: bad import: no-such-name is not exported by the import set

(import)
