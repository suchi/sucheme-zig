# Slice 4f: `+ - * /` and the comparisons over the exact numbers of the tower: fixnums,
# bignums and ratios. Flonums are written and read in 4g and 4r, so the inexact
# combinations are tested in the unit tests of src/num/ops.zig and src/prim/arith.zig.
# R7RS 6.2.6.

# 割り算は、割り切れなければ有理数になる。割り切れれば整数である。
(/ 1 2)
1/2

(/ 6 4)
3/2

(/ 6 3)
2

(/ 0 5)
0

(/ -1 3)
-1/3

(/ 1 -3)
-1/3

(/ -6 -4)
3/2

# 1つなら逆数。2つ以上なら、最初の数を残りで続けて割る。R7RS 6.2.6 の例。
(/ 3)
1/3

(/ -2)
-1/2

(/ 3 4 5)
3/20

(/ 1 2 3)
1/6

# 有理数の四則は約分する。分母が1になれば整数になる。
(+ (/ 1 2) (/ 1 3))
5/6

(- (/ 1 2) (/ 1 3))
1/6

(+ (/ 1 3) (/ 2 3))
1

(- (/ 1 2) (/ 1 2))
0

(* (/ 2 3) (/ 3 2))
1

(* (/ 1 2) 4)
2

(* (/ -1 2) (/ -1 2))
1/4

(+ 1 (/ 1 2))
3/2

(- (/ 1 3))
-1/3

(+ (/ 1 3) (/ 1 3) (/ 1 3))
1

# bignum の割り算と、約分で fixnum に戻る結果。
(/ 1 (* 4294967296 4294967296))
1/18446744073709551616

(/ (* 4294967296 4294967296) (* 4294967296 4294967296))
1

(/ (* 4294967296 4294967296) 4)
4611686018427387904

(/ (* 4294967296 3) (* 4294967296 6))
1/2

# bignum が fixnum に収まる値に戻れば、fixnum の手続きに渡せる。
(quotient (- (+ 9223372036854775807 1) 1) 3)
3074457345618258602

(- (* 4294967296 4294967296) (* 4294967296 4294967296))
0

(+ (* -4294967296 2147483648) (* 4294967296 2147483648))
0

# `number?` は、塔のどの数にも真である。

#t

(number? 'a)
#f
