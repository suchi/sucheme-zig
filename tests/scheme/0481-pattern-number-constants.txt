# Slice 4j: the numbers that a `syntax-rules` pattern can hold as constants (R7RS 7.1.3, `<pattern datum>`).
# 定数のパターンは `equal?` で照合するので、bignum と ratnum は値で一致する。

(define-syntax kind (syntax-rules () ((_ 1.5) 'flonum) ((_ 1/2) 'ratnum) ((_ 100000000000000000000) 'bignum) ((_ 1+2i) 'compnum) ((_ x) 'other))) (list (kind 1.5) (kind 1/2) (kind 100000000000000000000) (kind 1+2i))
(flonum ratnum bignum compnum)

(define-syntax kind (syntax-rules () ((_ 1.5) 'flonum) ((_ 1/2) 'ratnum) ((_ 100000000000000000000) 'bignum) ((_ 1+2i) 'compnum) ((_ x) 'other))) (list (kind 1.25) (kind 1/3) (kind 100000000000000000001) (kind 1+3i) (kind 1.5+0.0i))
(other other other other other)

# 値の比較は正確さを区別する。
(define-syntax kind (syntax-rules () ((_ 2) 'exact) ((_ 2.0) 'inexact) ((_ x) 'other))) (list (kind 2) (kind 2.0) (kind 4/2))
(exact inexact exact)

(define-syntax kind (syntax-rules () ((_ 0.0) 'zero) ((_ -0.0) 'negative-zero) ((_ x) 'other))) (list (kind 0.0) (kind -0.0))
(zero negative-zero)

# 定数は、リストとベクタの中にも書ける。テンプレートに書いた数は、そのまま出力される。
(define-syntax pick (syntax-rules () ((_ (1/2 x)) x) ((_ #(1.5 x)) (quote (x 100000000000000000000))))) (list (pick (1/2 'a)) (pick #(1.5 b)))
(a (b 100000000000000000000))

# バイトベクタは `<pattern datum>` にない。
(define-syntax m (syntax-rules () ((_ #u8(1)) 1)))
error: bad syntax-rules: this datum cannot be a pattern
