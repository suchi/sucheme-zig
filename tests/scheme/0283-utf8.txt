# スライス 3e: `utf8->string` と `string->utf8`(R7RS 6.9)。`string->utf8` の範囲は文字で数え、
# `utf8->string` の範囲はバイトで数える。バイトベクタが妥当な UTF-8 でないとき(範囲が文字の途中で
# 切れるときを含む)は、誤りである。基数の接頭辞(`#x41`)は段階4で入るので、原文の例は 10 進で書く。
# 各ケースは新しいインタプリタで走るので、定義は使う式と同じ行に置く。

# 原文の例: `(utf8->string #u8(#x41))` は "A"、`(string->utf8 "λ")` は `#u8(#xCE #xBB)`。
(utf8->string #u8(65))
"A"

(string->utf8 "λ")
#u8(206 187)

(string->utf8 "")
#u8()

(utf8->string #u8())
""

(string->utf8 "abc")
#u8(97 98 99)

(utf8->string #u8(97 98 99))
"abc"

# 1バイトから4バイトまでの文字。
(string->utf8 "é")
#u8(195 169)

(string->utf8 "あ")
#u8(227 129 130)

(string->utf8 "𐀀")
#u8(240 144 128 128)

(utf8->string #u8(240 144 128 128))
"𐀀"

(string-length (utf8->string #u8(97 206 187 227 129 130 240 144 128 128)))
4

# 往復する。
(utf8->string (string->utf8 "aλあ𐀀z"))
"aλあ𐀀z"

(string->utf8 (utf8->string #u8(97 206 187 227 129 130 240 144 128 128)))
#u8(97 206 187 227 129 130 240 144 128 128)

(equal? (utf8->string (string->utf8 "")) "")
#t

# 範囲。`string->utf8` は文字の位置、`utf8->string` はバイトの位置である。
(string->utf8 "aλあ𐀀z" 1 3)
#u8(206 187 227 129 130)

(string->utf8 "aλあ𐀀z" 2)
#u8(227 129 130 240 144 128 128 122)

(string->utf8 "aλあ𐀀z" 5)
#u8()

(string->utf8 "aλあ𐀀z" 2 2)
#u8()

(utf8->string #u8(97 206 187 227 129 130) 1 6)
"λあ"

(utf8->string #u8(97 206 187 227 129 130) 3)
"あ"

(utf8->string #u8(97 206 187 227 129 130) 3 3)
""

(utf8->string #u8(97 98 99) 1 2)
"b"

# 結果は新しいオブジェクトで、定数ではない。リテラルから作っても書き換えられる。
(define s (utf8->string #u8(97 98))) (string-set! s 0 #\z) s
"zb"

(define b (string->utf8 "ab")) (bytevector-u8-set! b 0 99) b
#u8(99 98)

(define b '#u8(97 98)) (define s (utf8->string b)) (bytevector-u8-set! (bytevector-copy b) 0 0) s
"ab"

# 妥当でない UTF-8 は断る。サロゲート、冗長な符号化、範囲外の値、途中で切れた並び、先頭でないバイト。

error: 1:1: argument out of range in utf8->string #u8(237 160 128)

(utf8->string #u8(244 144 128 128))
error: 1:1: argument out of range in utf8->string #u8(244 144 128 128)

(utf8->string #u8(206))
error: 1:1: argument out of range in utf8->string #u8(206)

# 範囲が文字の途中で切れる。
(utf8->string #u8(97 206 187 98) 2)
error: 1:1: argument out of range in utf8->string #u8(97 206 187 98) 2

(utf8->string #u8(97 206 187 98) 0 2)
error: 1:1: argument out of range in utf8->string #u8(97 206 187 98) 0 2

(utf8->string #u8(97 206 187 98) 1 3)
"λ"

# 型の誤り。型が、範囲より先に報告される。
(utf8->string "abc")
error: 1:1: wrong type of argument to utf8->string "abc"

(string->utf8 #u8(97))
error: 1:1: wrong type of argument to string->utf8 #u8(97)

(utf8->string '(97))
