# Slice 1k, unit K1: the pairs and the list the programs make. `cons` makes a
# pair, `car` and `cdr` take it apart, `pair?` and `null?` ask what a value is,
# and `list` makes the pairs of a list in one call.

# cons makes a pair, and a list is pairs that end in ().
(cons 1 2)
(1 . 2)

(cons 1 '())
(1)

(cons 1 (cons 2 '()))
(1 2)

(cons 1 (cons 2 3))
(1 2 . 3)

(cons '(1) '(2))
((1) 2)

(cons 'a "s")
(a . "s")

# Any value can be either half.
(cons #t #\a)
(#t . #\a)

(cons #(1) #(2))
(#(1) . #(2))

# car and cdr take a pair apart.
(car (cons 1 2))

(cdr (cons 1 2))

(cdr (cdr (cdr (list 1 2 3))))
()

(cdr '(1 . 2))

# Programs of these procedures and define, lambda and if. Each case is one line.
(define (append2 a b) (if (null? a) b (cons (car a) (append2 (cdr a) b)))) (append2 (list 1 2) (list 3 4))
(1 2 3 4)

(define (append2 a b) (if (null? a) b (cons (car a) (append2 (cdr a) b)))) (append2 '() '(1))
(1)

(define (append2 a b) (if (null? a) b (cons (car a) (append2 (cdr a) b)))) (append2 '(1) '())
(1)

(define (rev xs acc) (if (null? xs) acc (rev (cdr xs) (cons (car xs) acc)))) (rev (list 1 2 3) '())
(3 2 1)

(define (map1 f xs) (if (null? xs) '() (cons (f (car xs)) (map1 f (cdr xs))))) (map1 (lambda (x) (* x x)) (list 1 2 3 4))
(1 4 9 16)

(define (len xs) (if (null? xs) 0 (+ 1 (len (cdr xs))))) (len (list 1 2 3))
3

# A list of a few hundred pairs, made and read while the collector runs. The
# runner also runs every case with a collection at each step. The long-list
# scale test is planned in docs/plans/0010 (K6) and is not here yet.
(define (build n acc) (if (= n 0) acc (build (- n 1) (cons n acc)))) (define (sum xs acc) (if (null? xs) acc (sum (cdr xs) (+ acc (car xs))))) (sum (build 500 '()) 0)
125250

(define keep (list 1 2 3)) (define (spin n) (if (= n 0) 'ok (spin (- n 1)))) (spin 20000) keep

# list builds its pairs in a call whose arguments are themselves made by calls.
(define (iota n) (if (= n 0) '() (cons n (iota (- n 1))))) (list (iota 3) (iota 2) (iota 1) (iota 0))
