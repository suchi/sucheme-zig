# Slice 4g-4: `number->string` and `string->number` (R7RS 6.2.7). The reader does not read
# flonums or ratios yet (slice 4r), so those numbers are made by `/` and `inexact`.

# number->string は、基数10と、2、8、16で書く。
(list (number->string 255) (number->string 255 16) (number->string 255 2) (number->string 255 8) (number->string -255 16))
("255" "ff" "11111111" "377" "-ff")

(list (number->string (/ 1 3)) (number->string (/ 1 3) 2) (number->string (/ -7 16) 16))
("1/3" "1/11" "-7/10")

(list (number->string (inexact (/ 1 2))) (number->string (inexact 100)) (number->string (inexact (/ 1 3))))
("0.5" "100.0" "0.3333333333333333")

# 不正確な数を基数10以外で書くと、正確な有理数の綴りに `#i` が付く。
(number->string (inexact (/ 1 2)) 2)
"#i1/10"

(number->string (* 4294967296 4294967296))
"18446744073709551616"

(number->string (* 4294967296 4294967296) 16)
"10000000000000000"

# 基数が2、8、10、16でなければ、範囲の誤りである。
(number->string 10 3)
error: argument out of range in number->string 10 3

(number->string 10 'a)
error: wrong type of argument to number->string 10 a

(number->string 'a)
error: wrong type of argument to number->string a

# 無限大と NaN は、基数10以外では書き戻せない。
(number->string (/ (inexact 1) (inexact 0)) 16)
error: argument out of range in number->string +inf.0 16

(number->string (/ (inexact 1) (inexact 0)))
"+inf.0"

# string->number は、数の綴りを数にし、数でなければ #f を返す。
(list (string->number "100") (string->number "100" 16) (string->number "#o177" 2) (string->number "-ff" 16) (string->number "FF" 16))
(100 256 127 -255 255)

(list (string->number "1/3") (string->number "1e2") (string->number "#e0.1") (string->number "+inf.0") (string->number "1/0"))
(1/3 100.0 1/10 +inf.0 #f)

(list (string->number "") (string->number "abc") (string->number " 1") (string->number "1 ") (string->number "-") (string->number "#b2"))
(#f #f #f #f #f #f)

(string->number "12" 2)
#f

(string->number "18446744073709551616")
18446744073709551616

(string->number "10" 3)
error: argument out of range in string->number "10" 3

(string->number 10)
error: wrong type of argument to string->number 10

# 綴りが長すぎる数は #f である(桁数の上限)。
(string->number (make-string 100001 #\1))
#f

# 往復: 書いた綴りを読むと、同じ数になる(正確さも同じ)。bignum と ratnum の `eqv?` は 4j まで
# 同一のオブジェクトで比べるので、`=` と `exact?` で比べる。
(let ((n (/ -7 16))) (let ((m (string->number (number->string n 2) 2))) (list (= n m) (exact? m))))
(#t #t)

(let ((x (inexact (/ 1 10)))) (list (eqv? x (string->number (number->string x))) (eqv? x (string->number (number->string x 2) 2))))
(#t #t)

(let ((n (* 4294967296 4294967296 4294967296))) (let ((m (string->number (number->string n 8) 8))) (list (= n m) (exact? m))))
(#t #t)
