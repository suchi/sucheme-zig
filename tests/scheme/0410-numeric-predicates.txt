# Slice 4g-1: the type predicates of the numeric tower and the conversion between exact and
# inexact numbers (R7RS 6.2.6). The reader does not read flonums or ratios yet (slice 4r), so
# the numbers are made by `/` and `inexact`.

# 型の述語は、数でない値にも使えて、偽を返す。
(list (number? 1) (complex? 1) (real? 1) (rational? 1) (integer? 1) (exact-integer? 1))
(#t #t #t #t #t #t)

(list (complex? 'a) (real? 'a) (rational? 'a) (integer? 'a) (exact-integer? 'a) (complex? "1"))
(#f #f #f #f #f #f)

# R7RS 6.2.6 の例。
(list (rational? (/ 6 10)) (rational? (/ 6 3)) (integer? (/ 8 4)) (integer? (/ 1 2)) (exact-integer? (/ 32 5)))
(#t #t #t #f #f)

(list (exact-integer? 32) (exact-integer? (* 4294967296 4294967296)) (exact-integer? (inexact 32)))
(#t #t #f)

(list (integer? (* 4294967296 4294967296)) (integer? (/ (* 4294967296 4294967296) 3)) (rational? (* 4294967296 4294967296)))
(#t #f #t)

(list (integer? (inexact 3)) (integer? (inexact (/ 7 2))) (rational? (inexact (/ 7 2))) (real? (inexact 3)))
(#t #f #t #t)

# 正確かどうか。
(list (exact? 3) (exact? (/ 1 3)) (exact? (* 4294967296 4294967296)) (exact? (inexact 3)))
(#t #t #t #f)

(list (inexact? 3) (inexact? (/ 1 3)) (inexact? (inexact 3)) (inexact? (inexact (/ 1 3))))
(#f #f #t #t)

(exact? 'a)
error: 1:1: wrong type of argument to exact? a

(inexact? "1")
error: 1:1: wrong type of argument to inexact? "1"

# exact と inexact。
(exact 5)
5

(exact (/ 1 2))
1/2

(inexact 5)
5.0

(inexact (/ 1 3))
0.3333333333333333

(inexact (* 4294967296 4294967296))
18446744073709552000.0

(exact (inexact (/ 1 2)))
1/2

(exact (inexact (/ 3 4)))
3/4

(exact (inexact 5))
5

(list (exact 1) (exact (inexact (/ -9 2))) (exact (inexact (* 4294967296 4294967296))))
(1 -9/2 18446744073709551616)

(exact 'a)
error: 1:1: wrong type of argument to exact a

(inexact "1")
error: 1:1: wrong type of argument to inexact "1"

# 無限大と NaN には正確な値が無い。
(exact (/ (inexact 1) (inexact 0)))
error: 1:1: argument out of range in exact +inf.0

(exact (/ (inexact 0) (inexact 0)))
error: 1:1: argument out of range in exact +nan.0
