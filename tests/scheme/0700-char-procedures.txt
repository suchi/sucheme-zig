# R7RS 6.6: ギリシャ文字のアルファは Alphabetic で、小文字である。

(import (scheme char)) (list (char-upper-case? #\A) (char-upper-case? #\a) (char-lower-case? #\a) (char-lower-case? #\A) (char-upper-case? #\3) (char-lower-case? #\3))

# 文字だが、大文字でも小文字でもない文字(R7RS 6.6)。ひらがなと、二重字の Lt。

# 写像。

# 印字は ASCII の外の文字をそのまま書くので、以下の結果は `char->integer` の値(十進数)
# で比べる。コメントに十六進数を添える。
(import (scheme char)) (map char->integer (list (char-upcase #\x03B1) (char-downcase #\x0391) (char-upcase #\x0436) (char-downcase #\x0416)))
(913 945 1046 1078)
# 913 = #x391, 945 = #x3B1, 1046 = #x416, 1078 = #x436

# 単純写像を持たない文字は、そのまま返る。ß に単純な大文字は無い。
(import (scheme char)) (map char->integer (list (char-upcase #\x00DF) (char-upcase #\x3042) (char-downcase #\x3042)))
(223 12354 12354)
# 223 = #xDF, 12354 = #x3042

# 語末のシグマ。大文字と畳み込みでシグマに戻る。小文字は小文字のまま。
(import (scheme char)) (map char->integer (list (char-upcase #\x03C2) (char-foldcase #\x03C2) (char-downcase #\x03C2) (char-downcase #\x03A3)))
(931 963 962 963)
# 931 = #x3A3, 963 = #x3C3, 962 = #x3C2

# İ の単純な小文字は i。単純な畳み込みは無い。
(import (scheme char)) (map char->integer (list (char-downcase #\x0130) (char-foldcase #\x0130)))
(105 304)
# 105 = #\i, 304 = #x130

# 畳み込み。
(import (scheme char)) (map char->integer (list (char-foldcase #\A) (char-foldcase #\a) (char-foldcase #\x1E9E) (char-foldcase #\x01C5)))

# 223 = #xDF, 454 = #x1C6
