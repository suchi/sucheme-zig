# スライス 3e: バイトベクタの基本の手続き(R7RS 6.9)。`bytevector?` `make-bytevector` `bytevector`
# `bytevector-length` `bytevector-u8-ref` `bytevector-u8-set!` と `bytevector-append`。
# 検査の順序は、引数の個数、型、定数かどうか、値の範囲である。バイトは 0 から 255 までの
# 整数で、それ以外の値は、整数でも型の誤りである。
# 各ケースは新しいインタプリタで走るので、定義は使う式と同じ行に置く。

# 型の述語。
(bytevector? #u8(1 2 3))
#t

(bytevector? #u8())
#t

(bytevector? (bytevector))
#t

(bytevector? (make-bytevector 2 0))
#t

(bytevector? #(1 2 3))
#f

(bytevector? '(1 2 3))
#f

(bytevector? "abc")
#f

(bytevector? 1)
#f

(bytevector? '())
#f

(bytevector? 'a)
#f

# `make-bytevector`: 原文の例と、省略した場合(0 で埋める)。
(make-bytevector 2 12)
#u8(12 12)

(make-bytevector 3)
#u8(0 0 0)

(make-bytevector 0)
#u8()

(make-bytevector 0 12)
#u8()

(make-bytevector 2 255)
#u8(255 255)

(make-bytevector 2 0)
#u8(0 0)

# 呼ぶたびに新しいバイトベクタである。
(eq? (make-bytevector 1 1) (make-bytevector 1 1))
#f

(make-bytevector -1)
error: argument out of range in make-bytevector -1

(make-bytevector -1 0)
error: argument out of range in make-bytevector -1 0

(make-bytevector 1 256)
error: wrong type of argument to make-bytevector 1 256

(make-bytevector 1 -1)
error: wrong type of argument to make-bytevector 1 -1

(make-bytevector 1 #\a)
error: wrong type of argument to make-bytevector 1 #\a

(make-bytevector 1 "a")
error: wrong type of argument to make-bytevector 1 "a"

(make-bytevector #\a)
error: wrong type of argument to make-bytevector #\a

(make-bytevector "3")
error: wrong type of argument to make-bytevector "3"

# 型の誤りが、範囲の誤りより先に報告される。
(make-bytevector -1 256)
error: wrong type of argument to make-bytevector -1 256

(make-bytevector)
error: wrong number of arguments to make-bytevector

(make-bytevector 1 2 3)
error: wrong number of arguments to make-bytevector 1 2 3

(bytevector)

(bytevector 0 255)

# 手続きが作るバイトベクタは `equal?` でリテラルと等しい。
(equal? (bytevector 1 2 3) #u8(1 2 3))
#t

(equal? (make-bytevector 2 7) #u8(7 7))
#t

# `bytevector-u8-set!`: 原文の例。
(import (sucheme derived)) (let ((bv (bytevector 1 2 3 4))) (bytevector-u8-set! bv 1 3) bv)
#u8(1 3 3 4)

(define b (make-bytevector 3 0)) (bytevector-u8-set! b 0 255) (bytevector-u8-set! b 2 7) b
#u8(255 0 7)

# 値は返さない。
(bytevector-u8-set! (make-bytevector 1 0) 0 1)
<no output>

# 書き込みは同じオブジェクトに見える。
(define b (bytevector 1 2 3)) (define c b) (bytevector-u8-set! c 0 9) b
#u8(9 2 3)

(bytevector-u8-set! (bytevector 1 2 3) 3 0)

# `bytevector-append`: 原文の例。
(bytevector-append #u8(0 1 2) #u8(3 4 5))
#u8(0 1 2 3 4 5)

(bytevector-append)
#u8()

(bytevector-append #u8())
#u8()

(bytevector-append #u8() #u8())
#u8()

(bytevector-append #u8(1) #u8() #u8(2 3) #u8())
#u8(1 2 3)

# 引数が1つでも、結果は新しいバイトベクタである。
(define a (bytevector 1 2)) (define c (bytevector-append a)) (bytevector-u8-set! c 0 9) (list a c)
(#u8(1 2) #u8(9 2))

(define a (bytevector 1 2)) (eq? a (bytevector-append a))
#f

# 同じ引数を何度並べてもよい。
(define a (bytevector 1 2)) (bytevector-append a a a)
#u8(1 2 1 2 1 2)

(bytevector-append #u8(1) "a")
error: wrong type of argument to bytevector-append #u8(1) "a"

(bytevector-append '() #u8(1))
error: wrong type of argument to bytevector-append () #u8(1)

(bytevector-append #(1) #u8(1))
error: wrong type of argument to bytevector-append #(1) #u8(1)
