# スライス 3e: ベクタのリテラルは定数で(R7RS 3.4 と 4.1.2)、書き込む手続きは誤りとして断る。
# 検査の順序は、型、定数かどうか、範囲である。範囲が空の書き込みも、定数なら断る。
# 手続きが作るベクタは定数ではない。バイトベクタの対応するケースは 0284 にある。
# 各ケースは新しいインタプリタで走るので、定義は使う式と同じ行に置く。

# 原文(R7RS 6.8)の例: `(vector-set! '#(0 1 2) 1 "doe")` は、定数のベクタで誤りである。
(vector-set! '#(0 1 2) 1 "doe")
error: cannot modify a constant in vector-set! #(0 1 2) 1 "doe"

(vector-set! #(0 1 2) 1 "doe")
error: cannot modify a constant in vector-set! #(0 1 2) 1 "doe"

(vector-set! (quote #(0 1 2)) 1 0)
error: cannot modify a constant in vector-set! #(0 1 2) 1 0

# 変数が定数を指していても、定数である。
(define v #(1 2 3)) (vector-set! v 0 9)
error: cannot modify a constant in vector-set! #(1 2 3) 0 9

# 手続きの本体のリテラルは、呼ぶたびに同じ定数である。
(define (g) #(1 2 3)) (vector-set! (g) 0 9)
error: cannot modify a constant in vector-set! #(1 2 3) 0 9

(define (g) '#(1 2 3)) (eq? (g) (g))
#t

# 空の定数も断る。
(vector-set! #() 0 0)
error: cannot modify a constant in vector-set! #() 0 0

# 検査の順序は、型、定数、範囲である。
(vector-set! #(1 2 3) 99 0)
error: cannot modify a constant in vector-set! #(1 2 3) 99 0

(vector-set! #(1 2 3) -1 0)
error: cannot modify a constant in vector-set! #(1 2 3) -1 0

(vector-set! #(1 2 3) #\a 0)
error: wrong type of argument to vector-set! #(1 2 3) #\a 0

# 入れ子のリテラルの中のベクタも定数である。
(vector-set! (car '(#(1))) 0 9)
error: cannot modify a constant in vector-set! #(1) 0 9

(vector-set! (vector-ref #(#(1)) 0) 0 9)
error: cannot modify a constant in vector-set! #(1) 0 9

(vector-set! (cdr '(1 . #(1 2))) 0 9)
error: cannot modify a constant in vector-set! #(1 2) 0 9

(set-car! (vector-ref #((1 2)) 0) 9)
error: cannot modify a constant in set-car! (1 2) 9

# マクロが出力するリテラルも定数である。
(define-syntax m (syntax-rules () ((_) #(1 2)))) (vector-set! (m) 0 9)
error: cannot modify a constant in vector-set! #(1 2) 0 9

(define-syntax m (syntax-rules () ((_) '#(1 2)))) (vector-set! (m) 0 9)
error: cannot modify a constant in vector-set! #(1 2) 0 9

# 書き込めるのは、手続きが作ったベクタである。
(define v (vector 1 2 3)) (vector-set! v 0 9) v
#(9 2 3)

(define v (make-vector 2 0)) (vector-set! v 1 5) v
#(0 5)

(define v (vector-append #(1) #(2))) (vector-set! v 0 9) v
#(9 2)

(define v (list->vector '(1 2))) (vector-set! v 0 9) v
#(9 2)

# 定数の写しは定数ではない(原文の例)。
(define v (vector-copy #(1 2 3))) (vector-set! v 0 9) v
#(9 2 3)

(define v (vector-append #(1 2 3))) (vector-set! v 0 9) v
#(9 2 3)

(define v (vector-copy #(1 2 3) 1)) (vector-fill! v 0) v
#(0 0)

# 段階3の文字列からの変換が作るベクタも、定数ではない。
(define v (string->vector "ab")) (vector-set! v 0 #\z) v
#(#\z #\b)

# `vector-fill!`: 定数なら断る。何も書かない呼び出しも断る。
(vector-fill! #(1 2 3) 0)
error: cannot modify a constant in vector-fill! #(1 2 3) 0

(vector-fill! #(1 2 3) 0 1 1)
error: cannot modify a constant in vector-fill! #(1 2 3) 0 1 1

(vector-fill! #() 0)
error: cannot modify a constant in vector-fill! #() 0

(vector-fill! #(1 2 3) 0 9)
error: cannot modify a constant in vector-fill! #(1 2 3) 0 9

(vector-fill! #(1 2 3) 0 0 9)
error: cannot modify a constant in vector-fill! #(1 2 3) 0 0 9

(vector-fill! #(1 2 3) 0 #\a)
error: wrong type of argument to vector-fill! #(1 2 3) 0 #\a

# `vector-copy!`: 先が定数なら断る。元が定数なのはかまわない。
(vector-copy! #(1 2 3) 0 #(9))
error: cannot modify a constant in vector-copy! #(1 2 3) 0 #(9)

(vector-copy! #(1 2 3) 0 (vector 9))
error: cannot modify a constant in vector-copy! #(1 2 3) 0 #(9)

(vector-copy! #(1 2 3) 0 #())
error: cannot modify a constant in vector-copy! #(1 2 3) 0 #()

(vector-copy! #(1 2 3) 0 #(9) 1 1)
error: cannot modify a constant in vector-copy! #(1 2 3) 0 #(9) 1 1

(vector-copy! #() 0 #())
error: cannot modify a constant in vector-copy! #() 0 #()

(vector-copy! #(1 2 3) 99 #(9))
error: cannot modify a constant in vector-copy! #(1 2 3) 99 #(9)

(vector-copy! #(1 2 3) 0 #(9 9) 0 5)
error: cannot modify a constant in vector-copy! #(1 2 3) 0 #(9 9) 0 5

(vector-copy! #(1 2 3) 0 '(9))
error: wrong type of argument to vector-copy! #(1 2 3) 0 (9)

(define v (make-vector 3 0)) (vector-copy! v 0 #(7 8 9)) v
#(7 8 9)

# 定数を読むのはかまわない。
(define v #(1 2 3)) (list (vector-ref v 0) (vector-length v) (vector-copy v 1) (vector->list v) (equal? v #(1 2 3)))
(1 3 #(2 3) (1 2 3) #t)

# 先が定数でも、start と end の型が合わなければ、型の誤りが先に出る(型、不変、範囲の順)。
(vector-copy! #(1 2 3) 0 (vector 1) 'a)
error: wrong type of argument to vector-copy! #(1 2 3) 0 #(1) a

(vector-copy! #(1 2 3) 0 (vector 1) 0 'a)
error: wrong type of argument to vector-copy! #(1 2 3) 0 #(1) 0 a

# 書き込みを断られても、定数はそのままである。
(define v #(1 2 3)) (call-with-current-continuation (lambda (k) (with-exception-handler (lambda (e) (k v)) (lambda () (vector-fill! v 0)))))
#(1 2 3)

# 定数の中のペアも、これまでどおり定数である。
(set-car! (vector-ref #((1 2)) 0) 0)
error: cannot modify a constant in set-car! (1 2) 0

# 手続きの `quote` を通らない `list` のペアは、ベクタに入れても書き換えられる。
(define p (list 1 2)) (define v (vector p)) (set-car! (vector-ref v 0) 9) p
(9 2)

# 準引用が作るベクタは、リテラルではないので定数ではない。
(import (sucheme derived)) (define v `#(1 ,(+ 1 1))) (vector-set! v 0 9) v
#(9 2)
