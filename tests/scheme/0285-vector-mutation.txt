# スライス 3e: ベクタの生成と変更(R7RS 6.8)。`make-vector` `vector-set!` `vector-fill!` と、
# リストとの変換 `vector->list` `list->vector`。検査の順序は、引数の個数、型、定数かどうか、
# 値の範囲である。範囲は、始まりを含み、終わりを含まない。
# 各ケースは新しいインタプリタで走るので、定義は使う式と同じ行に置く。

# `make-vector`: 原文の説明どおり、k 個の要素を持つ。`fill` がなければ、#f で埋める。
(make-vector 3 0)
#(0 0 0)

(make-vector 2 'a)
#(a a)

(make-vector 0)
#()

(make-vector 0 'a)
#()

(make-vector 2)
#(#f #f)

(vector-length (make-vector 1000 0))
1000

# 要素は同じオブジェクトである(写しを作らない)。
(define v (make-vector 2 (vector 1))) (eq? (vector-ref v 0) (vector-ref v 1))
#t

# 呼ぶたびに新しいベクタである。
(eq? (make-vector 1 0) (make-vector 1 0))
#f

# `vector-set!`: 原文の例。
(define vec (vector 0 '(2 2 2 2) "Anna")) (vector-set! vec 1 '("Sue" "Sue")) vec
#(0 ("Sue" "Sue") "Anna")

# 値は返さない。
(vector-set! (vector 1 2) 0 9)
<no output>

(define v (make-vector 3 0)) (vector-set! v 0 1) (vector-set! v 2 'x) v
#(1 0 x)

# どんな値も入れられる。ベクタ自身も入れられる。
(define v (vector 1 2)) (vector-set! v 0 v) (eq? v (vector-ref v 0))
#t

(define v (vector 1 2)) (vector-set! v 1 '()) v
#(1 ())

# `list->vector`: 原文の例。
(list->vector '(dididit dah))
#(dididit dah)

(list->vector '())
#()

(list->vector '(1))
#(1)

(list->vector (list 1 2 3))
#(1 2 3)

(list->vector '((1 2) #(3) "x"))
#((1 2) #(3) "x")

# 往復する。
(vector->list (list->vector '(1 2 3)))
(1 2 3)

(list->vector (vector->list #(1 2 3)))
#(1 2 3)

# 結果は新しいベクタで、定数ではない。リテラルのリストから作っても書き換えられる。
(define v (list->vector '(1 2 3))) (vector-set! v 0 9) v
#(9 2 3)

# 先頭の部分だけが同じベクタを2つ作ったときも、別のオブジェクトである。

# 輪になったリストは、リストではない。誤りを捕まえれば、止まる。
(define l (list 1 2 3)) (set-cdr! (cdr (cdr l)) l) (call-with-current-continuation (lambda (k) (with-exception-handler (lambda (e) (k 'rejected)) (lambda () (list->vector l)))))
rejected

# `vector-fill!`: 原文の例。
(define a (vector 1 2 3 4 5)) (vector-fill! a 'smash 2 4) a
#(1 2 smash smash 5)

(define a (vector 1 2 3)) (vector-fill! a 0) a
#(0 0 0)

(define a (vector 1 2 3)) (vector-fill! a 0 1) a
#(1 0 0)

(define a (vector 1 2 3)) (vector-fill! a 0 3) a
#(1 2 3)

(define a (vector 1 2 3)) (vector-fill! a 0 1 1) a
#(1 2 3)

(define a (vector 1 2 3)) (vector-fill! a 0 0 3) a
#(0 0 0)

(vector-fill! (vector) 0)
<no output>

(vector-fill! (vector 1 2) 0)
<no output>
