# Slice 1g at a scale. These cases take most of the time of a run, so they run
# with `zig build test-scheme -Dslow=true` and not on every commit
# (docs/plans/0007-test-tiers.md). Each has a smaller twin in
# tests/scheme/0012-gc.txt, or in the unit tests of src/machine.zig, that runs
# the same logic.

# A million iterations of a tail call, in memory that does not depend on the count.
(define (loop n) (if (= n 0) 'done (loop (- n 1)))) (loop 1000000)
done

# A non-tail recursion that never returns runs into the limit of the heap and
# reports it, instead of using up the memory of the process.
(define (f n) (+ 1 (f n))) (f 0)
error: heap limit exceeded

# The deep recursion of slice 1f still returns, a hundred thousand levels deep.
(define (count n) (if (= n 0) 0 (+ 1 (count (- n 1))))) (count 100000)
100000
